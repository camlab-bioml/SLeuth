    def apply_gene_correction(gene):
        """Apply gene name correction from GENE_CORRECTIONS dictionary
        Uses case-insensitive lookup and returns uppercase result"""
        if pd.isna(gene):
            return gene
        gene_upper = str(gene).upper()
        # Lookup in uppercase version of corrections
        return GENE_CORRECTIONS_UPPER.get(gene_upper, gene_upper)

    def load_cv_folds(directory, model_name, cv_label, verify_genes=None):
        """Load cross-validation folds for a single model.

        Args:
            directory: Directory containing fold predictions
            model_name: Name for error messages
            cv_label: Label in filename (e.g., 'cv1', 'cv2', 'cv3')
            verify_genes: Expected gene list to verify against (None for first model)

        Returns:
            tuple: (stacked numpy array, gene_names list)
        """

        current_genes = None

        for i in range(N_CV_FOLDS):
            df = pd.read_csv(f'{OUTPUT_DIR}/{directory}/fold_{i}_{cv_label}_predictions.csv', index_col=0)
            files.append(df.values)

            if i == 0:
                current_genes = df.index.tolist()
                # Verify row and column names match (critical for symmetric matrix interpretation)
                assert df.columns.tolist() == current_genes, f"{model_name}: Row and column names don't match!"

                # Verify consistency with previously loaded gene names if provided
                if verify_genes is not None:
                    assert current_genes == verify_genes, f"{model_name}: Gene order inconsistent with previous models!"

        return np.stack(files), current_genes

            SL_score_matrix[idx1, idx2] = score

    SL_status_matrix = (SL_score_matrix > SL_THRESHOLD).astype(float)

    # Calculate naive edge probability estimate using vectorized NumPy operations
    # For each gene, calculate probability = (# of SL connections) / (# of valid genes)
    # This properly handles NaN values by excluding them from both numerator and denominator
    print("\n  Calculating naive edge probability estimate...")

    # Create working copy and set diagonal to NaN to exclude self-connections
    SL_status_no_diag = SL_status_matrix.copy()
    np.fill_diagonal(SL_status_no_diag, np.nan)

    # Vectorized calculation: for each row, sum non-NaN values / count non-NaN values
    # nansum: sum ignoring NaN (numerator = # of SL connections)
    # count_nonzero(~isnan): count of valid (non-NaN) entries (denominator = # of valid genes)
    row_sums = np.nansum(SL_status_no_diag, axis=1)
    row_counts = np.count_nonzero(~np.isnan(SL_status_no_diag), axis=1)

    # Calculate probability vector: avoid division by zero
    prob_vector = np.where(row_counts > 0, row_sums / row_counts, 0)

    # Create naive edge probability matrix via outer product
    naive_edge_est_matrix = np.outer(prob_vector, prob_vector)

    # Verify symmetry (outer product of vector with itself is always symmetric)
    assert np.allclose(naive_edge_est_matrix, naive_edge_est_matrix.T), "Naive edge estimate matrix not symmetric!"

# Bootstrap function for confidence intervals
def bootstrap_auc(y_true, y_pred, n_bootstraps=None, metric='roc'):
    """Calculate bootstrap confidence intervals for AUC metrics
    Uses global n_bootstrap parameter if not specified"""
    if n_bootstraps is None:
        n_bootstraps = n_bootstrap
    rng = np.random.RandomState(42)
    aucs = []
    n = len(y_true)

    for _ in range(n_bootstraps):
        # Sample with replacement
        indices = rng.choice(n, n, replace=True)
        y_true_boot = y_true.iloc[indices]
        y_pred_boot = y_pred.iloc[indices]

        if metric == 'roc':
            fpr, tpr, _ = roc_curve(y_true_boot, y_pred_boot)
            aucs.append(auc(fpr, tpr))
        elif metric == 'pr':
            aucs.append(average_precision_score(y_true_boot, y_pred_boot))

    return np.percentile(aucs, [2.5, 97.5])

# Bootstrap function for correlation confidence intervals
def bootstrap_correlation(x, y, n_bootstraps=None, method='pearson'):
    """Calculate bootstrap confidence intervals for correlations
    Uses global n_bootstrap parameter if not specified"""
    if n_bootstraps is None:
        n_bootstraps = n_bootstrap
    rng = np.random.RandomState(42)
    correlations = []
    n = len(x)

    for _ in range(n_bootstraps):
        indices = rng.choice(n, n, replace=True)
        x_boot = x.iloc[indices]
        y_boot = y.iloc[indices]

        if method == 'pearson':
            corr, _ = pearsonr(x_boot, y_boot)
        elif method == 'spearman':
            corr, _ = spearmanr(x_boot, y_boot)
        correlations.append(corr)

    return np.percentile(correlations, [2.5, 97.5])

def compute_metrics_with_ci(y_true, y_pred, n_bootstrap_iters=None):
    """Compute AUROC and AUPR with bootstrap confidence intervals.
    Uses global n_bootstrap parameter if not specified."""
    if n_bootstrap_iters is None:
        n_bootstrap_iters = n_bootstrap

    # Convert to numpy arrays for consistent indexing
    y_true_arr = np.asarray(y_true)
    y_pred_arr = np.asarray(y_pred)

    auroc = roc_auc_score(y_true_arr, y_pred_arr)
    aupr = average_precision_score(y_true_arr, y_pred_arr)
    print(f"    Computing metrics on {len(y_true_arr)} complete cases: AUROC={auroc:.4f}, AUPR={aupr:.4f}")

    def bootstrap_iteration(seed):
        np.random.seed(seed)
        idx = np.random.choice(len(y_true_arr), len(y_true_arr), replace=True)
        y_boot = y_true_arr[idx]
        p_boot = y_pred_arr[idx]
        return roc_auc_score(y_boot, p_boot), average_precision_score(y_boot, p_boot)

    results = Parallel(n_jobs=-1)(delayed(bootstrap_iteration)(i)
                                  for i in tqdm(range(n_bootstrap_iters), leave=False))
    auroc_scores, aupr_scores = zip(*results)

    auroc_ci = np.percentile(auroc_scores, [2.5, 97.5])
    aupr_ci = np.percentile(aupr_scores, [2.5, 97.5])

    return {'AUROC': auroc, 'AUROC_CI': auroc_ci,
            'AUPR': aupr, 'AUPR_CI': aupr_ci}

def determine_threshold(data, pred_col, value_col):
    """Determine optimal threshold direction using configured percentiles."""
    top_percentile, bot_percentile = THRESHOLD_PERCENTILES
    top_val = np.percentile(data[value_col], top_percentile)
    bot_val = np.percentile(data[value_col], bot_percentile)
    
    auroc_top = roc_auc_score((data[value_col] >= top_val).astype(int), data[pred_col])
    auroc_bot = roc_auc_score((data[value_col] <= bot_val).astype(int), data[pred_col])
    
    if auroc_top > auroc_bot:
        return 'top', top_val, (data[value_col] >= top_val).astype(int)
    return 'bottom', bot_val, (data[value_col] <= bot_val).astype(int)
