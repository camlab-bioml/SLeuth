from gene_corrections_config import GENE_CORRECTIONS

# Create case-insensitive lookup dictionary (uppercase keys)
GENE_CORRECTIONS_UPPER = {k.upper(): v.upper() for k, v in GENE_CORRECTIONS.items()}

def apply_gene_correction(gene):
    """Apply gene name correction from GENE_CORRECTIONS dictionary
    Uses case-insensitive lookup and returns uppercase result"""
    if pd.isna(gene):
        return gene
    gene_upper = str(gene).upper()
    # Lookup in uppercase version of corrections
    return GENE_CORRECTIONS_UPPER.get(gene_upper, gene_upper)

def load_cv_folds(directory, model_name, cv_label, verify_genes=None):
    """Load cross-validation folds for a single model.

    Args:
        directory: Directory containing fold predictions
        model_name: Name for error messages
        cv_label: Label in filename (e.g., 'cv1', 'cv2', 'cv3')
        verify_genes: Expected gene list to verify against (None for first model)

    Returns:
        tuple: (stacked numpy array, gene_names list)
    """
    files = []
    current_genes = None

    for i in range(N_CV_FOLDS):
        df = pd.read_csv(f'{OUTPUT_DIR}/{directory}/fold_{i}_{cv_label}_predictions.csv', index_col=0)
        files.append(df.values)

        if i == 0:
            current_genes = df.index.tolist()
            # Verify row and column names match (critical for symmetric matrix interpretation)
            assert df.columns.tolist() == current_genes, f"{model_name}: Row and column names don't match!"

            # Verify consistency with previously loaded gene names if provided
            if verify_genes is not None:
                assert current_genes == verify_genes, f"{model_name}: Gene order inconsistent with previous models!"

    return np.stack(files), current_genes

Step 3b: Calculate Naive Edge Probability Estimate

Code 
SL_status_no_diag = SL_status_matrix.copy()
np.fill_diagonal(SL_status_no_diag, np.nan)

row_sums = np.nansum(SL_status_no_diag, axis=1)
row_counts = np.count_nonzero(~np.isnan(SL_status_no_diag), axis=1)
prob_vector = np.where(row_counts > 0, row_sums / row_counts, 0)

naive_edge_est_matrix = np.outer(prob_vector, prob_vector)
assert np.allclose(naive_edge_est_matrix, naive_edge_est_matrix.T)

epsilon = 1e-10
naive_edge_est_logit_matrix = np.log(
    np.clip(naive_edge_est_matrix, epsilon, 1 - epsilon) /
    (1 - np.clip(naive_edge_est_matrix, epsilon, 1 - epsilon))
)
assert np.allclose(naive_edge_est_logit_matrix, naive_edge_est_logit_matrix.T)

Part 1: Per-Gene Logistic Regression

# Check if cached results exist
if os.path.exists('coefficients_per_gene_6xN.pkl'):
    print("Loading cached coefficients...")
    with open('coefficients_per_gene_6xN.pkl', 'rb') as f:
        coefficients_per_gene = pickle.load(f)
    print(f"Loaded coefficients: {coefficients_per_gene.shape}")
else:
    # Initialize 6×N matrix: 5 coefficients + 1 gene connecting probability
    coefficients_per_gene = np.full((6, n_genes), np.nan)

    # Store gene connecting probability as row 5
    coefficients_per_gene[5, :] = prob_vector

    # Fit per-gene logistic regression models
    for i in tqdm(range(n_genes), desc="Fitting per-gene models"):
        try:
            y = tensor_7xNxN[2, i, :]
            X = tensor_7xNxN[[3, 4, 5, 6], i, :].T

            df = pd.DataFrame({
                'SL_status': y,
                'naive_edge_logit': X[:, 0],
                'GO_BP': X[:, 1],
                'GO_CC': X[:, 2],
                'PPI': X[:, 3]
            }).dropna()

            lr = LogisticRegression(fit_intercept=True, max_iter=1000)
            lr.fit(df[['naive_edge_logit', 'GO_BP', 'GO_CC', 'PPI']], df['SL_status'])

            coefficients_per_gene[:5, i] = np.concatenate([[lr.intercept_[0]], lr.coef_[0]])
        except:
            pass

    n_success = np.sum(~np.isnan(coefficients_per_gene[0, :]))
    print(f"\nFitted {n_success}/{n_genes} models successfully")

    with open('coefficients_per_gene_6xN.pkl', 'wb') as f:
        pickle.dump(coefficients_per_gene, f)

print("\nCoefficient summary:")

Part 4: Coefficient vs Gene Connecting Probability

from scipy.stats import pearsonr, spearmanr

fig, axes = plt.subplots(3, 2, figsize=(12, 10))
axes = axes.flatten()

gene_prob = coefficients_per_gene[5, :]

for idx, name in enumerate(['Intercept', 'Naive Edge Logit', 'GO_BP', 'GO_CC', 'PPI']):
    ax = axes[idx]
    coef = coefficients_per_gene[idx, :]
    valid_mask = ~(np.isnan(coef) | np.isnan(gene_prob))

    if np.sum(valid_mask) > 0:
        x = gene_prob[valid_mask]
        y = coef[valid_mask]

        ax.scatter(x, y, alpha=0.5, s=20, edgecolors='black', linewidth=0.5)
        ax.axhline(y=0, color='red', linestyle='--', linewidth=1, alpha=0.5)
        ax.set_xlabel('Gene Connecting Probability')
        ax.set_ylabel('Coefficient Value')
        ax.set_title(f'{name} (n={len(x)})', fontweight='bold')
        ax.grid(True, alpha=0.3)

        if len(x) > 1:
            r_pearson, p_pearson = pearsonr(x, y)
            r_spearman, p_spearman = spearmanr(x, y)
            corr_text = f'Pearson r: {r_pearson:.3f} (p={p_pearson:.2e})\nSpearman ρ: {r_spearman:.3f} (p={p_spearman:.2e})'
            ax.text(0.02, 0.98, corr_text, transform=ax.transAxes, va='top', fontsize=8,
                    bbox=dict(boxstyle='round', facecolor='lightblue', alpha=0.5))
    else:
        ax.text(0.5, 0.5, 'No valid data', transform=ax.transAxes, ha='center', va='center')
        ax.set_title(f'{name} (n=0)', fontweight='bold')

axes[5].remove()

Output

The final tensor tensor_7xNxN has shape (7, {n_genes}, {n_genes}) containing: - Layer 0: SLMGAE predictions (CV1 without ESM) - continuous values - Layer 1: SL scores - continuous values from reference data - Layer 2: SL status - binary (0=non-SL, 1=SL) - Layer 3: Naive edge probability estimate - logit-transformed probability scores - Layer 4: GO BP similarity - continuous similarity scores - Layer 5: GO CC similarity - continuous similarity scores - Layer 6: PPI network - binary/weighted edges

All layers are symmetric (N×N matrices where element [i,j] = element [j,i]).
